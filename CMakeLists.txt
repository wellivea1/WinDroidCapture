cmake_minimum_required(VERSION 3.24)
project(WinDroidCapture VERSION 0.3.0 LANGUAGES C)
option(AWM_NATIVE "Build app-owned output components" ON)
option(AWM_LEGACY "Build the archived OBS bridge and its tests" OFF)
if(AWM_NATIVE AND AWM_LEGACY)
  message(FATAL_ERROR "Select one controller: AWM_NATIVE=ON or AWM_LEGACY=ON, not both")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This project targets Windows x64")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 COBJMACROS _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  add_compile_options(/W4 /utf-8 /guard:cf /sdl)
  add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
endif()
enable_testing()
if(AWM_NATIVE)
  add_subdirectory(native)
endif()
add_library(awm_backend STATIC src/core.c src/json.c src/platform.c src/audio.c)
if(AWM_LEGACY)
  target_sources(awm_backend PRIVATE src/obs.c)
endif()
target_include_directories(awm_backend PUBLIC src)
if(NOT AWM_LEGACY)
  target_compile_definitions(awm_backend PRIVATE AWM_NATIVE_ONLY)
endif()
target_link_libraries(awm_backend PUBLIC ws2_32 bcrypt crypt32 ole32 uuid propsys shell32 shlwapi user32 version mfplat mf mfuuid strmiids)
if(AWM_LEGACY)
add_executable(WinDroidCapture WIN32 src/main.c src/engine.c src/app.rc)
set_property(SOURCE src/app.rc APPEND PROPERTY OBJECT_DEPENDS "${PROJECT_SOURCE_DIR}/src/assets/windroid-capture.ico")
target_link_options(WinDroidCapture PRIVATE /MANIFEST:NO)
target_link_libraries(WinDroidCapture PRIVATE awm_backend comctl32 comdlg32 gdi32)
add_executable(awm-diagnostics src/diagnostics.c)
target_link_libraries(awm-diagnostics PRIVATE awm_backend)
add_executable(awm-receiver tests/receiver.c)
target_link_libraries(awm-receiver PRIVATE awm_backend gdi32)
enable_testing()
add_executable(awm-tests tests/test_core.c)
target_link_libraries(awm-tests PRIVATE awm_backend)
add_test(NAME core-and-platform COMMAND awm-tests)
add_executable(awm-engine-tests tests/test_engine.c src/engine.c)
target_link_libraries(awm-engine-tests PRIVATE awm_backend)
add_test(NAME simulated-engine-lifecycle COMMAND awm-engine-tests)
install(TARGETS WinDroidCapture awm-diagnostics awm-receiver RUNTIME DESTINATION .)
endif()
